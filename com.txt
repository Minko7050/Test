library ieee;
use ieee.std_logic_1164.all;
entity comparator is
port(x,y:in std_logic_vector(3 downto 0);eq,gt,lt:out std_logic);
end comparator;
architecture behavior of comparator is
begin
process(x,y)
begin
if(x=y)then
eq<='1';
gt<='0';
lt<='0';
elsif(x<y)then
eq<='0';gt<='0';
lt<='1';
else
eq<='0';
gt<='1';
lt<='0';
end if;
end process;
end behavior;
