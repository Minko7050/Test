library ieee;
use ieee.std_logic_1164.all;
entity demux is
port (s:in std_logic_vector(1 downto 0);
x:in std_logic;
z:out std_logic_vector(3 downto 0));
end demux;
architecture behavior of demux is
begin 
process (s,x)
begin
case s is
when "00"=>z(0)<=x;z(1)<='0';z(2)<='0';z(3)<='0';
when "01"=>z(1)<=x;z(0)<='0';z(2)<='0';z(3)<='0';

when "10"=>z(2)<=x;z(0)<='0';z(1)<='0';z(3)<='0';
when others =>z(3)<=x;z(0)<='0';z(1)<='0';z(2)<='0';

end case;
end process;
end behavior;
