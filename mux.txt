library ieee;
use ieee.std_logic_1164.all;
entity mux is
port (x:in std_logic_vector(3 downto 0);
s:in std_logic_vector(1 downto 0);
z:out std_logic);
end mux;
architecture behavior of mux is
begin 
process (s,x)
begin case s is
when "00"=>z<=x(0);
when "01"=>z<=x(1);
when "10"=>z<=x(2);
when others =>z<=x(3);
end case;
end process;
end behavior;
