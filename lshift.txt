library ieee;
use ieee.std_logic_1164.all;
entity lshift is
port(clk,clr,din:in std_logic;q:buffer std_logic_vector(3 downto 0);dout:out std_logic);
end lshift;
architecture behavior of lshift is 
begin 
process(clk,clr)
begin
if(clr='1')then q<="0000";
elsif(rising_edge(clk))then

q(3 downto 1)<=q(2 downto 0);
dout<=q(3);
q(0)<=din;

end if;
end process;
end behavior;
