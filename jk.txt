library ieee;
use ieee.std_logic_1164.all;
entity JKflipflop is
  port ( jk:in std_logic_vector(1 downto 0);
         x,clk,clr :in std_logic;
         z:out std_logic);
end JKflipflop;
architecture behavior of JKflipflop is
begin 
  process(clk,clr)
    begin
      if(clr='1')then z<='0';
      elsif(clk'event and clk='1') or (rising_edge(clk)) then
      case jk is
      when "00"=>z<=x;
      when "01"=>z<='0';
      when "10"=>z<='1';
      when others=>z<=not x;
    end case;
    end if;
    end process;
  end behavior;
