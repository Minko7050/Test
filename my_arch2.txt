library ieee;
use ieee.std_logic_1164.all;
entity my_arch is
port(A,B,C,D :in std_logic;Y:out std_logic);
end my_arch;
architecture structure of my_arch is
signal alpha,beta,thida,gamma,pie,dema: std_logic;
component andgate port(e,f:in std_logic;g:out std_logic);
end component;
component orgate port(p,q:in std_logic;r:out std_logic);
end component;
component notgate port(t:in std_logic;w:out std_logic);
end component;
begin
NOT1:notgate port map(A,alpha);
NOT2:notgate port map(B,beta);
NOT3:notgate port map(C,thida);
NOT4:notgate port map(D,gamma);
AND1:andgate port map(alpha,beta,pie);

AND2:andgate port map(thida,gamma,dema);
OR1:orgate port map(pie,dema,Y);
end structure;
